cmake_minimum_required(VERSION 3.25)
project(connectcpp VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Protobuf CONFIG REQUIRED)
find_package(absl CONFIG REQUIRED)
find_package(llhttp CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
find_package(OpenSSL 3 REQUIRED)
find_path(ASIO_INCLUDE_DIR asio/awaitable.hpp REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(NGHTTP2 REQUIRED IMPORTED_TARGET libnghttp2>=1.61.0)
add_library(connectcpp src/envelope.cc src/streaming_protocol.cc src/message_stream.cc src/server_transport.cc src/client_transport.cc src/codec.cc src/http2.cc src/http2_stream.cc src/http2_connection.cc src/transport.cc src/tls.cc src/unary.cc src/rpc.cc)
add_library(connectcpp::connectcpp ALIAS connectcpp)
set(runtime_generated "${CMAKE_CURRENT_BINARY_DIR}/runtime_generated")
add_custom_command(
  OUTPUT "${runtime_generated}/connectrpc/internal/grpc_status.pb.cc"
         "${runtime_generated}/connectrpc/internal/grpc_status.pb.h"
  COMMAND ${CMAKE_COMMAND} -E make_directory "${runtime_generated}"
  COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto
    --cpp_out=${runtime_generated} connectrpc/internal/grpc_status.proto
  DEPENDS proto/connectrpc/internal/grpc_status.proto
  VERBATIM)
target_sources(connectcpp PRIVATE src/interceptor.cc src/http_connection.cc src/cors.cc src/get.cc src/http1.cc src/http1_connection.cc src/grpc_protocol.cc src/grpc_web_protocol.cc
  "${runtime_generated}/connectrpc/internal/grpc_status.pb.cc")
target_include_directories(connectcpp PRIVATE "${runtime_generated}")
target_compile_features(connectcpp PUBLIC cxx_std_20)
target_include_directories(connectcpp SYSTEM PUBLIC $<BUILD_INTERFACE:${ASIO_INCLUDE_DIR}>)
target_compile_definitions(connectcpp PUBLIC ASIO_STANDALONE)
# Use nghttp2's portable nghttp2_ssize APIs; MSVC has no POSIX ssize_t.
target_compile_definitions(connectcpp PRIVATE NGHTTP2_NO_SSIZE_T=1)
target_include_directories(connectcpp PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(connectcpp PUBLIC protobuf::libprotobuf OpenSSL::SSL PRIVATE llhttp::llhttp ZLIB::ZLIB PkgConfig::NGHTTP2 absl::strings)
set_target_properties(connectcpp PROPERTIES CXX_EXTENSIONS OFF)
option(CONNECTCPP_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CONNECTCPP_THREAD_SANITIZE "Enable the thread sanitizer separately" OFF)
if(CONNECTCPP_SANITIZE AND CONNECTCPP_THREAD_SANITIZE)
  message(FATAL_ERROR "ASan/UBSan and TSan require separate build directories")
endif()
if(MSVC AND (CONNECTCPP_SANITIZE OR CONNECTCPP_THREAD_SANITIZE))
  message(FATAL_ERROR "These sanitizer configurations require GCC or Clang")
endif()
option(CONNECTCPP_PROTOBUF_GLOBALS_COMPAT
  "Use Protobuf's non-ASan message globals layout with a prebuilt runtime" OFF)
if(CONNECTCPP_PROTOBUF_GLOBALS_COMPAT)
  # Protobuf 36 enables a different generated globals representation under
  # ASan. Use its upstream opt-out when linking a non-instrumented runtime.
  target_compile_definitions(connectcpp PUBLIC PROTOBUF_MESSAGE_GLOBALS_TEMPORARY_OPTOUT=1)
endif()
if(CONNECTCPP_SANITIZE AND NOT MSVC)
  target_compile_options(connectcpp PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(connectcpp PUBLIC -fsanitize=address,undefined)
elseif(CONNECTCPP_THREAD_SANITIZE)
  target_compile_options(connectcpp PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(connectcpp PUBLIC -fsanitize=thread)
endif()
if(MSVC)
  target_compile_options(connectcpp PRIVATE /W4)
else()
  target_compile_options(connectcpp PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  include(cmake/GoogleTest.cmake)
  add_executable(connectcpp_test tests/protocol_test.cc tests/streaming_protocol_test.cc tests/http2_test.cc tests/http2_stream_test.cc tests/http2_connection_test.cc tests/transport_test.cc tests/tls_test.cc tests/unary_test.cc)
  include(cmake/TestCertificate.cmake)
  target_link_libraries(connectcpp_test PRIVATE connectcpp GTest::gtest_main)
  target_sources(connectcpp_test PRIVATE tests/interceptor_test.cc tests/callback_test.cc tests/http_connection_test.cc tests/cors_test.cc tests/get_test.cc tests/http1_test.cc tests/http1_connection_test.cc tests/grpc_protocol_test.cc tests/grpc_web_protocol_test.cc)
  include(GoogleTest)
  gtest_discover_tests(connectcpp_test NO_PRETTY_VALUES PROPERTIES TIMEOUT 15)
endif()
install(TARGETS connectcpp EXPORT connectcppTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT connectcppTargets NAMESPACE connectcpp:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/connectcpp)
configure_package_config_file(cmake/connectcppConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/connectcppConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/connectcpp)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/connectcppConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/connectcppConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/connectcppConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/connectcpp)
option(CONNECTCPP_BUILD_GENERATOR "Build the Go protoc plugin" ${BUILD_TESTING})
if(CONNECTCPP_BUILD_GENERATOR)
  include(cmake/Codegen.cmake)
endif()
option(CONNECTCPP_BUILD_CONFORMANCE "Build and run the pinned official conformance suite" OFF)
if(CONNECTCPP_BUILD_CONFORMANCE)
  include(cmake/Conformance.cmake)
endif()
option(CONNECTCPP_BUILD_FUZZERS "Build libFuzzer protocol targets (Clang)" OFF)
if(CONNECTCPP_BUILD_FUZZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT CONNECTCPP_SANITIZE)
    message(FATAL_ERROR "Fuzzing requires Clang and CONNECTCPP_SANITIZE=ON")
  endif()
  target_compile_options(connectcpp PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(connectcpp_protocol_fuzz fuzz/protocol_fuzz.cc)
  target_link_libraries(connectcpp_protocol_fuzz PRIVATE connectcpp)
  target_compile_options(connectcpp_protocol_fuzz PRIVATE -fsanitize=fuzzer-no-link)
  target_link_options(connectcpp_protocol_fuzz PRIVATE -fsanitize=fuzzer)
endif()
option(CONNECTCPP_BUILD_BENCHMARKS "Build the optional gRPC C++ comparison" OFF)
if(CONNECTCPP_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()
